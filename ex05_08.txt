// 사운드의 진폭 변화를 원의 크기 변화로 표현

let soundFile, myAmp;

function preload() {
  soundFile = loadSound('data/latin.mp3'); 	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  createCanvas(500, 500);
  soundFile.play();				// 사운드 파일 재생
  myAmp = new p5.Amplitude();		// p5.Amplitude 객체(myAmp) 생성
}

function draw() {
  background(0, 10);				// 페이딩 효과
  let level = myAmp.getLevel();		// 사운드 샘플 진폭값(0∼1) 구하기 
  let myLevel = map(level, 0, 1, 10, 500);	// 사운드 샘플 진폭값 범위 변경

  noStroke();					// 원의 외곽선 제거 (부드러운 페이딩 효과)
  fill(255, 255, 0);
  ellipse(width/2, height/2, myLevel, myLevel);	// 사운드 샘플 진폭값 = 원의 크기
}
