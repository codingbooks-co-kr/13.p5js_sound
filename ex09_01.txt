// 440Hz 삼각파 오실레이터의 주파수 스펙트럼 표시

let osc, fft;

function setup() {
  let canvas = createCanvas(500, 200);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누를 때 canvasPressed() 호출

  osc = new p5.Oscillator();			// p5.Oscillator 객체(osc) 생성
  osc.setType('triangle');			// 오실레이터의 유형 설정(삼각파)
  osc.freq(440);					// 오실레이터의 초기 주파수 설정(440Hz)
  fft = new p5.FFT();				// 스펙트럼 표시를 위해 p5.FFT 객체(fft) 생성
}

function draw() {
  background(0);
  let spectrum = fft.analyze();				// 주파수 영역의 진폭값 배열 반환
  stroke(0, 255, 0);
  for (let i = 0; i < spectrum.length; i++) {		// 기본값: spectrum.length=1024
    let x = map(i, 0, spectrum.length, 0, width);	// x: 주파수
    let y = map(spectrum[i], 0, 255, height, 0);	// y: 진폭
    line(x, height, x, y);				// 주파수 스펙트럼을 선으로 표시
  }

  noStroke();
  fill(255);
  textSize(20);
  text('파형 타입: ' + osc.getType(), 10, 30);
}

function canvasPressed() {			// 콜백함수
  osc.start();					// 오실레이터 시작
}

function mouseReleased() {			// 이벤트함수
  osc.stop();					// 오실레이터 정지
}
