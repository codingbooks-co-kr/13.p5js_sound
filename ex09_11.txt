// 마이크 입력 신호의 스펙트로그램

let mic, fft, bins = 128, x = 0, step = 3;

function setup() {
  createCanvas(500, 200);
  mic = new p5.AudioIn();			// p5.AudioIn 객체(mic) 생성
  mic.start();					// 마이크 입력 시작
  fft = new p5.FFT(0.8, bins);			// 0.8: 스무딩값, bins: 스펙트럼 배열 크기
  fft.setInput(mic);				// fft 객체의 입력소스(→mic 객체) 설정
  
  background(0);
  colorMode(HSB);				// HSB 컬러모드
}

function draw() {
  let spectrum = fft.analyze();			// 주파수 영역의 스펙트럼 분석
  noStroke();
  for (i = 0; i < spectrum.length; i++) {	// spectrum.length=128 (=bins값)
    let y = map(i, 0, spectrum.length, height, 0);
    let rectHeight = height / spectrum.length;
    let h = map(spectrum[i], 0, 255, 0, 240);	// Hue값 변환: 진폭(0∼255)→색상(0o∼240o)
    fill(h, 100, 100);				// S(채도)와 B(밝기)가 100%일 때 색상(H) 설정
    rect(x, y-rectHeight, step, rectHeight);
  }
  x = (x + step) % width;			// x=0,3,6,9,...,498 순으로 3씩 증가를 반복
}
