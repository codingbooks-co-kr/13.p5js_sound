// 사운드 파일 재생 시 시간 파형의 직선형 시각화-2
// 사운드 샘플들을 선으로 연결하여 파형 그리기

let soundFile, myAmp, samples = [];

function preload() {
  soundFile = loadSound('data/latin.mp3');	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  createCanvas(500, 200);
  myAmp = new p5.Amplitude();		// p5.Amplitude 객체(myAmp) 생성
  soundFile.loop();				// 사운드 파일 무한 반복 재생
  for (let i = 0; i < width; i++) {		// 배열 초기화, 배열의 크기는 width(=500)
    samples[i] = height;			// 진폭값: 화면 최하단(height)에 표시
  }
}

function draw() {
  background(0);
  let level = myAmp.getLevel();		// 사운드 샘플 진폭값(0∼1) 구하기
  let myLevel = map(level, 0, 0.5, height, 0);	// 실제 사용된 진폭값 범위: 0∼0.5

  for (let i = 0; i < width-1; i++) {
    samples[i] = samples[i+1];			// (1) 배열요소의 복사
  }
  samples[width-1] = myLevel;		// 새로운 진폭값을 배열 끝에 저장
  
  stroke(255);
  for (let i = 0; i < width-1; i++) {
    line(i, samples[i], i+1, samples[i+1]);	// (2) 진폭값들을 선으로 연결하기
  }
}
