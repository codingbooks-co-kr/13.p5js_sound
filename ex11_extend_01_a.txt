// 사운드 파일 재생 시 시간 파형의 직선형 시각화-1
// 사운드 샘플 꼭지점들을 연결하여 파형 그리기

let soundFile, myAmp, samples = [];

function preload() {
  soundFile = loadSound('data/latin.mp3');	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  createCanvas(500, 200);
  soundFile.loop();				// 사운드 파일 무한 반복 재생
  myAmp = new p5.Amplitude();		// p5.Amplitude 객체(myAmp) 생성
}

function draw() {
  background(0);
  let level = myAmp.getLevel();		// 사운드 샘플 진폭값(0∼1) 구하기
  let myLevel = map(level, 0, 0.5, height, 0);	// 실제 사용된 진폭값 범위: 0∼0.5
  samples.push(myLevel);			// samples 배열 끝에 진폭값(myLevel) 추가
  
  stroke(255);					// 파형 선 색상
  noFill();					// 꼭지점 연결 도형의 채움 색상 없음
  beginShape();					// 시간 파형 그리기 시작
  for (let i = 0; i < samples.length; i++) {
    vertex(i, samples[i]);			// 꼭지점
  }
  endShape();					// 시간 파형 그리기 종료
  
  if (samples.length > width) {			// samples 배열의 크기가 width보다 클 때
    samples.splice(0, 1);			// 첫 번째 배열요소인 samples[0] 삭제
  }
  stroke(255, 0, 0);
  line(samples.length, 0, samples.length, height);	// 진행선 그리기
}
