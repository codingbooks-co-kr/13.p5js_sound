// 마이크 입력 신호의 시간 파형을 회전형 진폭선으로 표시-1

let mic, theta = 0;

function setup() {
  createCanvas(500, 500);  
  mic = new p5.AudioIn();		// p5.AudioIn 객체(mic) 생성
  mic.start();				// 마이크 입력 시작
}

function draw() {
  // ❶ 회전형 검은색 선으로 이전값 지우기
  let x = width/2 + width * cos(theta);
  let y = height/2 + height * sin(theta);
  stroke(0);
  strokeWeight(5);
  line(width/2, height/2, x, y);

  // ❷ 회전형 진폭선으로 마이크 입력 신호의 크기를 표현
  let level = mic.getLevel();				// 마이크 음량(진폭: 0∼1) 구하기
  let myLevel = map(level, 0, 0.5, 0, width);		// 실제 사용된 진폭값 범위: 0∼0.5
  x = width/2 + myLevel * cos(theta);		// 원주상의 x좌표
  y = height/2 + myLevel * sin(theta);		// 원주상의 y좌표
  stroke(0, 255, 0);					// 회전형 진폭선 색상 설정
  line(width/2, height/2, x, y);				// 회전형 진폭선 그리기

  theta += 0.01;					// 회전 속도
  if (theta > TWO_PI) theta = 0;			// theta는 0∼2π 범위를 반복
}
