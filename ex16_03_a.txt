// 아두이노: 주파수값 수신 후 부저 제어

#define BUZZER 13

void setup() {
  pinMode(BUZZER, OUTPUT);
  Serial.begin(9600);
  while (Serial.available() == 0) {	// 수신 버퍼에 수신값이 없는 동안 반복 실행
    Serial.write('A');			// 시작신호('A') 전송
    delay(300);				// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {			// 이벤트함수: 수신값이 있으면 자동 호출
  int myFreq = Serial.readStringUntil('\n').toInt();	// 수신 (문자열에서 주파수 추출)
  if (myFreq == 0) {
    noTone(BUZZER);			// 무음
  } else {
    tone(BUZZER, myFreq);		// 수신 주파수음 출력
  }
  Serial.write('B');        		// 송신 (통신제어신호: 'B')
}
