// 노이즈 객체와 필터링 객체 연결-2

let noise, myFilter;

function setup() {
  let canvas = createCanvas(400, 100);
  canvas.mousePressed(canvasPressed);

  myFilter = new p5.Filter('lowpass');

  noise = new p5.Noise('white');
  noise.disconnect();		// 연결해제
  noise.connect(myFilter);	// 연결
  myFilter.process(noise, 500);

  background(200);
  textSize(30);
  textAlign(CENTER, CENTER);
  text('노이즈 필터링(LPF)', width/2, height/2);
}

function canvasPressed() {
  noise.start();
}

function mouseReleased() {
  noise.stop();
}
