// millis()를 사용한 음 순차 재생 (polySynth 대신 monoSynth도 가능)

let polySynth, index = 0, interval = 0, current_time = 0, previous_time = 0;
let sclaes = [60, 62, 64, 65, 67, 69, 71, 72];		// 미디번호배열(C장조음계)

function setup() {
  createCanvas(500, 200);
  polySynth = new p5.PolySynth();		// p5.PolySynth 객체(polySynth) 생성

  fill(255);
  textSize(30);
  textAlign(CENTER, CENTER);
}

function draw() {
  current_time = millis();			// 현재 시간 설정
  interval = current_time - previous_time;	// 시간차(interval) = 현재시간 - 이전시간
  if (interval > 500) {				// 500msec마다 if문은 true가 됨
    previous_time = millis();			// 500msec마다 현재 시간 설정
    let myFreq = midiToFreq(sclaes[index]);		// 미디번호를 주파수로 변환
    polySynth.play(myFreq, 1, 0, 0.3);			// 음 재생

    background(0);
    text('미디번호: ' + sclaes[index] + ', 주파수: ' + int(myFreq) + 'Hz', width/2, height/2);
    index = (index + 1) % sclaes.length;		// index = 0∼7 순으로 반복
  }
}
