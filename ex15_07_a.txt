// 아두이노: 진폭값 및 색상값 수신 후 LED 밝기 및 RGB LED 색상 제어, 버튼값 2개 송신

#define BUTTON1 2
#define BUTTON2 4
#define LED 6
#define RedLED 11
#define GreenLED 5
#define BlueLED 3

void setup() {
  pinMode(BUTTON1, INPUT);
  pinMode(BUTTON2, INPUT);
  pinMode(LED, OUTPUT);
  pinMode(RedLED, OUTPUT);
  pinMode(GreenLED, OUTPUT);
  pinMode(BlueLED, OUTPUT);
  Serial.begin(9600);
  while (Serial.available() == 0) {	// 수신 버퍼에 수신값이 없는 동안 반복 실행
    Serial.println('A');			// 시작신호('A') 전송(→문자+개행 전송)
    delay(300);				// 0.3초 대기
  }
}

void loop() { }

void serialEvent() {				// 이벤트함수: 수신값이 있으면 자동 호출
  int myLevel = Serial.readStringUntil(',').toInt();	// 수신 (진폭값, 음량값)
  int myColor = Serial.readStringUntil('\n').toInt();	// 수신 (색상값)

  analogWrite(LED, myLevel);			// 수신값(=진폭값)으로 LED 밝기 제어
  if (myColor == 1) setColor(255, 255, 255);	// 수신값(=색상값)으로 색상 제어: White 
  else if (myColor == 2) setColor(255, 0, 0);	// 수신값(=색상값)으로 색상 제어: Red
  else if (myColor == 3) setColor(0, 255, 0);	// 수신값(=색상값)으로 색상 제어: Green
  else if (myColor == 4) setColor(0, 0, 255);	// 수신값(=색상값)으로 색상 제어: Blue

  int value1 = digitalRead(BUTTON1);		// 버튼값1 읽기
  int value2 = digitalRead(BUTTON2);		// 버튼값2 읽기
  Serial.print(value1);
  Serial.print(",");
  Serial.println(value2);				// 송신 (문자열: “value1,value2\n“)
}

void setColor(int Red, int Green, int Blue) {	// 함수: 색상 결정
  digitalWrite(RedLED, Red);
  digitalWrite(GreenLED, Green);
  digitalWrite(BlueLED, Blue);
}
