// 캔버스를 눌러 세 개의 음 동시 재생

let polySynth;

function setup() {
  let canvas = createCanvas(500, 200);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누를 때 canvasPressed() 호출
  polySynth = new p5.PolySynth();		// p5.PolySynth 객체(polySynth) 생성 
  
  background(200);
  textSize(30);
  textAlign(CENTER, CENTER);
  text('캔버스를 눌러 세 개의 음 동시 재생', width/2, height/2);
}

function canvasPressed() {			// 콜백함수: 세 개의 음을 동시 재생
  polySynth.play(midiToFreq(60), 1, 0, 0.3);	// 미디번호 60('C4')의 주파수: 262Hz
  polySynth.play(midiToFreq(64), 1, 0, 0.3);	// 미디번호 64('E4')의 주파수: 330Hz
  polySynth.play(midiToFreq(67), 1, 0, 0.3);	// 미디번호 67('G4')의 주파수: 392Hz
}
