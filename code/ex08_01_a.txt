// 회전운동하는 작은 원-1

let theta = 0, radius = 100;

function setup() {
  createCanvas(300, 300);
}

function draw() {
  background(0);
  stroke(255);
  noFill();
  ellipse(width/2, height/2, radius*2, radius*2);

  let x = width/2 + radius * cos(theta);
  let y = height/2 + radius * sin(theta);
  line(width/2, height/2, x, y);

  fill(0, 255, 0);
  ellipse(x, y, 20, 20);
  
  theta += 0.05;
}
