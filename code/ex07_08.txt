// 캔버스를 눌러 세 개의 랜덤 음 동시 재생

let polySynth;
let sclaes = [60, 62, 64, 65, 67, 69, 71];	// 미디번호 배열
let octaves = [3, 4, 5, 6, 7];			// 옥타브 배열

function setup() {
  let canvas = createCanvas(500, 200);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누를 때 canvasPressed() 호출
  polySynth = new p5.PolySynth();		// p5.PolySynth 객체(polySynth) 생성

  background(200);
  textSize(30);
  textAlign(CENTER, CENTER);
  text('캔버스를 눌러 랜덤 화음 생성', width/2, height/2);
}

function canvasPressed() {			// 콜백함수
  background(random(255), random(255), random(255));
  text('캔버스를 눌러 랜덤 화음 생성', width/2, height/2);
  for (let i = 0; i < 3; i++) {
    let note = random(sclaes);			// 미디번호 배열요소(60∼72) 중 랜덤 선택
    let octave = random(octaves);		// 옥타브 배열요소(3∼7) 중 랜덤 선택
    if (octave === 3) note -= 12;		// 3옥타브 (12음계를 빼면 1옥타브 하강)
    else if (octave === 5) note += 12;	// 5옥타브일 경우
    else if (octave === 6) note += 24;	// 6옥타브일 경우
    else if (octave === 7) note += 36;	// 7옥타브일 경우
    let myFreq = midiToFreq(note);		// 미디번호를 주파수로 변환
    polySynth.play(myFreq, 1, 0, 0.3);		// 음 재생
    console.log(i + " " + note + " " + int(myFreq) + 'Hz');
  }
}
