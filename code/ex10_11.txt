// 사운드 파일 재생 시 캔버스상에서 마우스의 좌우 위치에 따라 필터링효과 변화

let soundFile, myFilter;

function preload() {
  soundFile = loadSound('data/latin.mp3');	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  let canvas = createCanvas(500, 200);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누를 때 canvasPressed() 호출

  soundFile.disconnect();			// soundFile 객체의 출력 연결 해제
  soundFile.play();				// 사운드 파일 재생

  myFilter = new p5.Filter();			// p5.Filter 객체(myFilter) 생성
  myFilter.connect();				// myFilter 객체를 메인출력(→스피커)에 연결
  myFilter.setType('lowpass');			// 필터 유형 설정
  myFilter.process(soundFile, 500);		// 음원:soundFile, 필터링주파수:500Hz
}

function draw() {
  background(0);
  let x = constrain(mouseX, 0, width);
  stroke(255, 255, 0);
  line(x, 0, x, height);				// 마우스를 따라오는 선

  let pos = map(x, 0, width, 0, 1);
  myFilter.drywet(pos);				// pos=0∼1 (0:필터링없음, 1:필터링최대)
  
  noStroke();
  fill(255);
  textSize(20);
  text('마우스의 좌우 위치에 따라 필터링(LFP)효과 변화', 10, 30);
  text('Dry|Wet: ' + int(pos * 100) + '%', 10, 60);
}

function canvasPressed() {		// 콜백함수
  soundFile.playMode('restart');		// 이전 재생을 중지하고 다시 시작하는 모드
  soundFile.play();			// 사운드 파일 재생
}
