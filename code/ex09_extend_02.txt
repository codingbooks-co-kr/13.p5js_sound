// 주파수 스펙트럼 표시 (버튼으로 정지/시작 토글, 오실레이터 유형 선택)

let osc, fft, d, myToggle = false;
let button1, button2, button3, button4, button5;
    
function setup() {
  createCanvas(500, 200);
  osc = new p5.Oscillator();		// p5.Oscillator 객체(osc) 생성
  osc.setType('sine');			// 오실레이터의 유형 설정(기본값: 'sine')
  osc.freq(440);				// 오실레이터의 초기 주파수 설정(440Hz)
  osc.start();				// 오실레이터 시작
  
  fft = new p5.FFT();			// 스펙트럼 표시를 위해 p5.FFT 객체(fft) 생성
  
  d = createDiv('');			// 브라우저 화면에서 텍스트 출력(초기 텍스트 없음)
  d.style('font-size', '20px');		// 텍스트 폰트 크기를 (20픽셀로) 설정
  
  button1 = createButton('정지');	// 버튼 생성 (버튼 레이블은 '정지')
  button1.size(100, 50);		// 버튼 크기 설정
  button1.style('font-size', '20px');	// 버튼 폰트 크기를 (20픽셀로) 설정
  button1.mousePressed(toggleOsc);	// 버튼 누르면 toggleOsc() 자동 호출
  
  button2 = createButton('사인파');	// 버튼 생성 (버튼 레이블은 '사인파')
  button2.size(100, 50);		// 버튼 크기 설정
  button2.style('font-size', '20px');	// 버튼 폰트 크기를 (20픽셀로) 설정
  button2.mousePressed(setSine);	// 버튼 누르면 setSine() 자동 호출
  
  button3 = createButton('삼각파');	// 버튼 생성 (버튼 레이블은 '삼각파')
  button3.size(100, 50);		// 버튼 크기 설정
  button3.style('font-size', '20px');	// 버튼 폰트 크기를 (20픽셀로) 설정
  button3.mousePressed(setTriangle);	// 버튼 누르면 setTriangle() 자동 호출
  
  button4 = createButton('톱니파');	// 버튼 생성 (버튼 레이블은 '톱니파')
  button4.size(100, 50);		// 버튼 크기 설정
  button4.style('font-size', '20px');	// 버튼 폰트 크기를 (20픽셀로) 설정
  button4.mousePressed(setSawtooth);	// 버튼 누르면 setSawtooth() 자동 호출
  
  button5 = createButton('구형파');	// 버튼 생성 (버튼 레이블은 '구형파')
  button5.size(100, 50);		// 버튼 크기 설정
  button5.style('font-size', '20px');	// 버튼 폰트 크기를 (20픽셀로) 설정
  button5.mousePressed(setSquare);	// 버튼 누르면 setSquare() 자동 호출
}

function draw() {
  background(0);
  d.html('파형타입: ' + osc.getType());
  
  let spectrum = fft.analyze();			// 주파수 영역의 진폭값 배열 반환
  stroke(0, 255, 0);
  for (let i = 0; i < spectrum.length; i++) {	// 기본값: spectrum.length=1024
    let x = map(i, 0, spectrum.length, 0, width);	// x: 주파수
    let y = map(spectrum[i], 0, 255, height, 0);	// y: 진폭
    line(x, height, x, y);				// 주파수 스펙트럼을 선으로 표시
  }
}

function toggleOsc() {
  if (myToggle) {
    osc.start();				// 오실레이터 시작
    button1.html('정지');
  } else {
    osc.stop();				// 오실레이터 정지
    button1.html('시작');
  }
  myToggle = !myToggle;
}

function setSine() {
  osc.setType('sine');			// 오실레이터의 유형 설정: 사인파
}

function setTriangle() {
  osc.setType('triangle');		// 오실레이터의 유형 설정: 삼각파
}

function setSawtooth() {
  osc.setType('sawtooth');		// 오실레이터의 유형 설정: 톱니파
}

function setSquare() {
  osc.setType('square');			// 오실레이터의 유형 설정: 구형파
}
