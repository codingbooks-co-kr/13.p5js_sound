// 방법3❷: 캔버스를 눌러 오디오 활성화

let soundFile;

function preload() {
  soundFile = loadSound('data/latin.mp3');
}

function setup() {
  let canvas = createCanvas(200, 100);
  canvas.mousePressed(userStartAudio);
  background(0);
  
  // 오디오 자동 재생 제한 시뮬레이션
  getAudioContext().suspend();

  // 실행 초기 사운드 파일 자동 재생
  soundFile.play();
}
