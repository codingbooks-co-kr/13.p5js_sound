// 사운드 파일 재생 시 캔버스상에서 마우스의 좌우 위치에 따라 왜곡효과 변화-1

let soundFile, distortion;

function preload() {
  soundFile = loadSound('data/latin.mp3');	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  let canvas = createCanvas(500, 200);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누를 때 canvasPressed() 호출

  soundFile.disconnect();			// soundFile 객체의 출력 연결 해제
  soundFile.play();				// 사운드 파일 재생

  distortion = new p5.Distortion();		// p5.Distortion 객체(distortion) 생성
  distortion.connect();				// distortion 객체를 메인출력(→스피커)에 연결
  distortion.process(soundFile, 0.2, '4x');	// 음원, 왜곡량: 0.2, 오버샘플링: '4x'
}

function draw() {
  background(0);
  let x = constrain(mouseX, 0, width);
  stroke(255, 255, 0);
  line(x, 0, x, height);				// 마우스를 따라오는 선

  let pos = map(x, 0, width, 0, 1);
  distortion.drywet(pos);			// pos=0∼1 (0:왜곡없음, 1:왜곡최대)
  
  noStroke();
  fill(255);
  textSize(20);
  text('마우스의 좌우 위치에 따라 왜곡효과 변화', 10, 30);
  text('Dry|Wet: ' + int(pos * 100) + '%', 10, 60);
}

function canvasPressed() {		// 콜백함수
  soundFile.playMode('restart');		// 이전 재생을 중지하고 다시 시작하는 모드
  soundFile.play();			// 사운드 파일 재생
}
