// 사운드 파일의 재생 시 시간 파형의 방사형 표시-1

let soundFile, fft;

function preload() {
  soundFile = loadSound('data/latin.mp3');	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  createCanvas(500, 500);
  soundFile.loop();				// 사운드 파일 무한 반복 재생
  fft = new p5.FFT();				// 파형 표시를 위해 p5.FFT 객체(fft) 생성
}

function draw() {
  background(0);
  let wave = fft.waveform();			// 시간 영역의 진폭값 배열 반환
  stroke(0, 255, 0);				// 방사형 파형 선 색상
  strokeWeight(3);				// 선 굵기 설정
  noFill();					// 꼭지점 연결 도형의 채움 색상 없음
  beginShape();					// 시간 파형(방사형) 그리기 시작
  for (let i = 0; i < wave.length; i++) {	// 기본값: wave.length=1024
    let theta = map(i, 0, wave.length, 0, TWO_PI);	// theta: 0∼2π
    let r = map(wave[i], -1, 1, 0, width/2);		// r: 진폭(반지름)
    let x = width/2 + r * cos(theta);			// 원주상의 x좌표
    let y = height/2 + r * sin(theta);			// 원주상의 y좌표
    vertex(x, y);					// 방사형 원주상의 꼭지점 좌표
  }
  endShape(CLOSE);				// 시간 파형 그리기 종료(시작과 끝 연결)
}
