// p5.js: 신경망 분류(Neural Network Classification)를 이용한 음 영역 분류
// 캔버스를 세 개의 음 영역으로 나누어 학습하고 마우스를 누른 위치에 분류된 주파수음 출력
// 분류된 주파수음으로 아두이노의 부저음 제어, 임의의 키를 누르면 무음

let port = new p5.WebSerial();		// 시리얼 통신 객체 생성
let model, monoSynth, d, slider, num, state = 'training';
let info = '1.마우스를 눌러 데이터 수집, 2.슬라이더로 데이터 변경';
let result = '0', myFreq = 0, numCluster = 3, notes = {1: 262, 2: 330, 3: 392};

function setup() {
  let canvas = createCanvas(500, 500);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누르면 canvasPressed() 호출
  monoSynth = new p5.MonoSynth();		// p5.MonoSynth 객체(monoSynth) 생성 

  port.getPorts(); 			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton); 	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort); 	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  port.on('data', serialEvent); 		// 데이터 수신: ‘data’ 이벤트 발생→serialEvent() 호출
  
  d = createDiv('');				// 브라우저 화면에서 텍스트 출력
  d.position(0, height+10);			// 텍스트 출력 위치 설정
  d.style('font-size', '20px');			// 텍스트 폰트 크기를 (20픽셀로) 설정
  
  slider = createSlider(1, numCluster, 1, 1);	// 슬라이더 생성(범위:1∼3, 초깃값:1, 스텝:1)
  slider.position(0, height+40);			// 슬라이더 위치 설정
  slider.size(width);				// 슬라이더 크기 설정
  
  let button1 = createButton('모델학습'); 	// 버튼 생성 (버튼 레이블은 '모델학습')
  button1.position(0, height+70); 		// 버튼 위치 설정
  button1.size(120, 50); 			// 버튼 크기 설정
  button1.style('font-size', '20px');		// 버튼 폰트 크기를 (20픽셀로) 설정
  button1.mousePressed(trainModel); 		// 버튼 누르면 trainModel() 자동 호출
  
  let button2 = createButton('모델저장'); 	// 버튼 생성 (버튼 레이블은 '모델저장')
  button2.position(130, height+70); 		// 버튼 위치 설정
  button2.size(120, 50); 			// 버튼 크기 설정
  button2.style('font-size', '20px');		// 버튼 폰트 크기를 (20픽셀로) 설정
  button2.mousePressed(saveModel); 		// 버튼 누르면 saveModel() 자동 호출

  let options = {				// 객체리터럴로 객체(options) 생성
    inputs: ['x', 'y'],				// 키(inputs): 값('x', 'y' 좌표 배열)
    outputs: ['number'],			// 키(outputs): 값('number' 배열)
    task: 'classification',			// 수행하려는 작업의 유형: 'classification'
    debug: false				// 학습 과정을 그래프로 시각화: true
  };
  model = ml5.neuralNetwork(options);	// neuralNetwork 객체(model) 생성

  background(200);
  colorMode(HSB);				// HSB 컬러모드
}

function draw() {
  d.html(info);					// 문자열(info) 출력
  num = slider.value();				// 슬라이더 값 읽기
}

function canvasPressed() {			// 콜백함수: 캔버스 누를 때 호출
  let inputs = { x: mouseX, y: mouseY };	// 입력 데이터 객체(inputs) 생성
  if (state === 'training') {			// 학습: 초기 실행 시 if문은 참
    let outputs = { number: str(num) };	// 출력 레이블 객체(outputs) 생성
    model.addData(inputs, outputs);		// model에 입·출력 데이터 객체 추가
    monoSynth.play(notes[num], 1, 0, 0.3);	// 모델 학습 시 입력 데이터의 음 출력
    
    fill(0);
    textSize(20);
    text(num, mouseX, mouseY);		// 입력 데이터를 숫자(num)로 표시
    info = '수집 데이터 번호: ' + num;
  } else if (state == 'classification') {		// 분류: 학습 종료 후 분류 시 if문은 참
    model.classify(inputs, getResult);		// 분류: inputs 주어지면 getResult() 자동 호출
  }
}

function trainModel() {				// 콜백함수: '모델학습' 버튼 누를 때 호출
  model.normalizeData();			// 데이터 정규화(0∼1)
  let options = { epochs: 200 };		// options 객체 생성 (학습횟수: 200회)
  model.train(options, training, finished);	// 신경망 학습(옵션, 콜백함수, 콜백함수)
}

function training(epoch, loss) {			// 콜백함수: 신경망 학습 중 호출
  info = '학습중: ' + epoch;
}

function finished() {				// 콜백함수: 신경망 학습 완료 시 호출
  info = '학습종료, 분류시작!';
  state = 'classification';			// 학습 종료 후 분류 시 state값 변경
}

function getResult(error, results) {		// 콜백함수: 신경망 학습완료 후 분류시 호출
  if (error) {					// 인식 오류 시
    console.log(error);				// 또는 console.error(error);
    return;					// 함수 실행 종료
  }
  //console.log(results);				// 신경망 분류 결과 출력
  result = results[0].label;			// 신뢰도가 가장 큰 결과(results[0])의 label
  myFreq = notes[result];			// 아두이노 전송 주파수
  info = '분류주파수: ' + myFreq + 'Hz (임의의 키를 누르면 무음)';
  monoSynth.play(myFreq, 1, 0, 0.3);		// 분류된 주파수음 출력

  let h = map(result, 1, numCluster, 0, 240);
  fill(h, 100, 100);				// S(채도)와 B(밝기)가 100%일 때 색상(H) 설정
  ellipse(mouseX, mouseY, 35, 35);		// 출력을 원으로 표시

  fill(0);
  textAlign(CENTER, CENTER);
  text(result, mouseX, mouseY);		// 신뢰도가 가장 큰 label로 표시
}

function saveModel() {				// 콜백함수: '모델저장' 버튼 누를 때 호출
  model.save();					// 학습 완료된 모델 저장
}

function keyPressed() {				// 이벤트함수: 임의의 키를 누르는 경우 호출
  myFreq = 0;					// 아두이노 전송 주파수: 0Hz
}

function serialEvent() {				// 콜백함수: 수신값이 있으면 자동 호출
  let signal = port.read();			// 수신 (통신시작신호('A'), 통신제어신호('B'))
  port.write(myFreq + '\n');			// 송신 (문자열: '주파수\n')
}

function makePortButton() {			// 포트 첫 연결 시 포트 선택 버튼 생성
  let portButton = createButton('포트 선택'); 	// 버튼 생성
  portButton.mousePressed(choosePort); 	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort(); 				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수
  port.open().then(initiateSerial); 		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
