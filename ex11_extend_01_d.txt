// 사운드 파일 재생 시 시간 파형의 회전형 시각화-1

let soundFile, myAmp, samples = [];

function preload() {
  soundFile = loadSound('data/latin.mp3');	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  createCanvas(500, 500);
  soundFile.loop();				// 사운드 파일 무한 반복 재생
  myAmp = new p5.Amplitude();		// p5.Amplitude 객체(myAmp) 생성
  angleMode(DEGREES);			// DEGREES 모드 설정 (0o∼360o 범위)
}

function draw() {
  background(0);
  let level = myAmp.getLevel();		// 사운드 샘플 진폭값(0∼1) 구하기
  samples.push(level);				// samples 배열 끝에 진폭값(level) 추가
  
  stroke(0, 255, 0);				// 회전형 진폭선 색상
  strokeWeight(5);
  noFill();					// 꼭지점 연결 도형의 채움 색상 없음
  beginShape();					// 회전형 파형 그리기 시작
  for (let i = 0; i < samples.length; i++) {
    let r = map(samples[i], 0, 1, 0, 1000);	// 음량(진폭)값을 가시적으로 증폭
    let x = width/2 + r * cos(i);		// 원주상의 x좌표
    let y = height/2 + r * sin(i);		// 원주상의 y좌표
    vertex(x, y);				// 회전형 원주상의 꼭지점: 흰색
  }
  endShape(CLOSE);				// 회전형 파형 그리기 종료(시작과 끝 연결)

  if (samples.length > 360) {			// 샘플수가 360 이상일 때(각도:360o 해당}
    samples.splice(0, 1);			// 첫 번째 배열요소인 samples[0] 삭제
  }
}
