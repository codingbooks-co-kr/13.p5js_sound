// 캔버스를 누를 때마다 마이크입력 & 사운드파일 토글, 각 음량을 파형으로 표시 

let soundFile, mic, fft, mode = 0;

function preload() {
  soundFile = loadSound('data/latin.mp3'); 	// soundFile 객체 생성 및 초기화(파일 로딩)
}

function setup() {
  createCanvas(500, 200);
  mic = new p5.AudioIn();		// p5.AudioIn 객체(mic) 생성
  mic.start();				// 마이크 입력 시작
  
  fft = new p5.FFT();			// 파형 표시를 위해 p5.FFT 객체(fft) 생성
  fft.setInput(mic);			// fft 객체의 입력소스(→mic 객체) 설정
}

function draw() {
  background(0);
  let wave = fft.waveform();		// 시간 영역의 진폭값 배열 반환

  stroke(255);				// 파형 선 색상
  noFill();				// 꼭지점 연결 도형의 채움 색상 없음
  beginShape();				// 시간 파형 그리기 시작
  for (let i = 0; i < wave.length; i++) {	// 기본값: wave.length=1024
    let x = map(i, 0, wave.length, 0, width);	// x: 시간
    let y = map(wave[i], -1, 1, height, 0);	// y: 진폭
    vertex(x, y);				// 꼭지점
  }
  endShape();				// 시간 파형 그리기 종료
}

function mousePressed() {		// 이벤트함수
  mode = (mode + 1) % 2;		// mode = 0,1,0,1... 순으로 반복
  if (mode === 0) {
    mic.start();				// 마이크 입력 시작
    soundFile.stop();			// 사운드 파일 재생 정지
    fft.setInput(mic);			// fft 객체의 입력소스(→mic 객체) 설정
  } else {
    mic.stop();				// 마이크 입력 중지
    soundFile.play();			// 사운드 파일 재생 시작
    fft.setInput(soundFile);		// fft 객체의 입력소스(→soundFile 객체) 설정
  }
}
