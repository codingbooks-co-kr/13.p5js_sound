// p5.js: 마우스를 눌러 생성한 원들을 학습, 군집화된 원에 마우스가 위치하면 해당 주파수 송신

let port = new p5.WebSerial();				// 시리얼 통신 객체 생성
let kmeans, monoSynth, d, slider, data = [], dd = 35, cluster = false;
let centroid = 0, myFreq = 0, numCluster = 3, notes = [262, 330, 392];

function setup() {
  let canvas = createCanvas(500, 500);
  canvas.mousePressed(canvasPressed);	// 캔버스를 누를 때 canvasPressed() 호출
  monoSynth = new p5.MonoSynth();		// p5.MonoSynth 객체(monoSynth) 생성

  port.getPorts(); 			// 포트 사용 가능 여부 확인
  port.on('noport', makePortButton); 	// 기존 선택한 포트가 없으면 makePortButton() 호출
  port.on('portavailable', openPort); 	// 사용 가능한 포트가 있으면 openPort() 자동 호출
  port.on('data', serialEvent); 		// 데이터 수신: ‘data’ 이벤트 발생→serialEvent() 호출

  d = createDiv('Cluster 개수: ' + numCluster);
  d.position(0, height+10);			// 텍스트 출력 위치 설정
  d.style('font-size', '20px');			// 텍스트 폰트 크기를 (20픽셀로) 설정

  slider = createSlider(1, numCluster, numCluster, 1);
  slider.position(200, height+15);		// 슬라이더 위치 설정
  slider.size(300);				// 슬라이더 크기 설정
  slider.input(changeCluster);			// 슬라이더 입력 시 changeCluster() 호출
  
  let button = createButton('Clustering');	// 버튼 생성 (버튼 레이블은 'Clustering')
  button.position(0, height+50); 		// 버튼 위치 설정
  button.size(120, 50);				// 버튼 크기 설정
  button.style('font-size', '20px');		// 버튼 폰트 크기를 (20픽셀로) 설정
  button.mousePressed(getCluster); 		// 버튼 누르면 getCluster() 자동 호출

  background(0);
  colorMode(HSB);				// HSB 컬러모드
}

function draw() {
  d.html('Cluster 개수: ' + numCluster);
  if (cluster) {					// clustering(), changeCluster()에서 값 변경
    for (i = 0; i < kmeans.dataset.length; i++) {	// 입력데이터(→원)의 개수만큼 반복
      centroid = kmeans.dataset[i].centroid;		// 세 그룹일 때 centroid = 0,1,2
      let datapointX = kmeans.dataset[i][0];		// 데이터(원)의 x좌표
      let datapointY = kmeans.dataset[i][1];		// 데이터(원)의 y좌표

      let h = map(centroid+1, 1, numCluster, 0, 240);
      if (numCluster === 1) h = 0;			// 클러스터가 하나이면 색상은 Red
      fill(h, 100, 100);					// HSB 중 S(채도)와 B(밝기)는 100%
      ellipse(datapointX, datapointY, dd, dd);		// 데이터를 원으로 표시	

      fill(0);
      textAlign(CENTER, CENTER);
      text(centroid+1, datapointX, datapointY);		// 각 원형 데이터의 centroid 표시
      
      let distance = dist(mouseX, mouseY, datapointX, datapointY);
      if (distance < dd/2) {				// 마우스가 원 내부에 있는 경우 
        myFreq = notes[centroid];			// 송신 주파수
        monoSynth.triggerAttack(myFreq, 1, 0);	// 해당 음 출력 시작
        d.html('음 주파수: ' + myFreq + 'Hz');
        break;						// for문 빠져나옴
      } else {						// 마우스가 원 외부에 있는 경우
        myFreq = 0;					// 송신 주파수(0→무음)
        centroid = 0;					// [수정] 예제 참조
        monoSynth.triggerRelease();			// 음 출력 종료
      }
    }
  }
}

function canvasPressed() {			// 콜백함수: 캔버스 누를 때 호출
  let inputs = { x: mouseX, y: mouseY };	// 입력 데이터 객체(inputs) 생성
  data.push(inputs);				// data 배열에 inputs 객체 추가
  fill(255);
  ellipse(mouseX, mouseY, dd, dd);		// 입력 데이터를 원으로 표시
}

function getCluster() {				// 콜백함수: 버튼 누를 때 호출
  let options = {				// 옵션 객체(options) 생성
    k: numCluster,				// 클러스터 수
    maxIter: 1200,
    threshold: 0.9
  };
  kmeans = ml5.kmeans(data, options, clustering);	// 객체 생성 및 clustering() 호출
}

function clustering() {				// 콜백함수: 객체 생성 시 호출
  cluster = true;
}

function changeCluster() {			// 콜백함수: 클러스터값(1∼3) 변경
  console.log('클러스터 개수를 바꾼 경우 Clustering 버튼을 다시 눌러주세요!');
  numCluster = slider.value();			// 슬라이더값 읽기
  cluster = false;
}

function serialEvent() {				// 콜백함수: 수신값이 있으면 자동 호출
  let signal = port.read();			// 수신 (통신시작신호('A'), 통신제어신호('B'))
  port.write(myFreq + '\n');			// 송신 (문자열: '주파수\n')
}

function makePortButton() {			// 포트 첫 연결 시 포트 선택 버튼 생성
  let portButton = createButton('포트 선택'); 	// 버튼 생성
  portButton.mousePressed(choosePort); 	// 버튼 누르면 choosePort() 자동 호출
  function choosePort() {			// 콜백함수
    port.requestPort(); 				// 시리얼 포트 선택창 오픈
  }
}

function openPort() {				// 콜백함수
  port.open().then(initiateSerial); 		// 포트를 오픈 후 initiateSerial() 자동 호출
  function initiateSerial() {			// 콜백함수
    console.log('포트 오픈');
  }
}
